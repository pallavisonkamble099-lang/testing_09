<script>
function loadSample(){
 document.getElementById('notice').value =
 "B.Tech First Year Notice. Programming practical will be held on 28 October 2026. Submit Assignment 2 by 30 October 2026. Students must bring their college ID card.";
 document.getElementById('question').value = "When is the programming practical?";
 document.getElementById('answer').textContent = "Sample notice loaded. Click Find Answer or Summarise Notice.";
}

function askNotice(){
 const text=document.getElementById('notice').value.trim();
 const q=document.getElementById('question').value.trim();
 const out=document.getElementById('answer');
 if(!text||!q){out.textContent="Please enter a notice and a question.";return;}
 const sentences=text.match(/[^.!?\n]+[.!?]?/g)||[text];
 const words=q.toLowerCase().split(/\W+/).filter(w=>w.length>3);
 const matches=sentences.filter(s=>words.some(w=>s.toLowerCase().includes(w)));
 out.textContent=matches.length
 ? "Possible answer from your notice:\n"+matches.join("\n")+"\n\nPlease verify this against the original notice."
 : "I couldn't find a clear matching answer in the pasted text. Check the original notice or try different keywords.";
}

function summarise(){
 const text=document.getElementById('notice').value.trim();
 document.getElementById('answer').textContent=text
 ? "Notice text:\n"+text+"\n\nReview the important dates and tasks above. This basic prototype does not yet generate an AI summary."
 : "Please paste a college notice first.";
}

function addTask(){
 const name=document.getElementById('taskName').value.trim();
 const date=document.getElementById('taskDate').value;
 if(!name||!date){alert("Please enter a task name and deadline.");return;}
 const item=document.createElement('div');
 item.className='task';
 const label=document.createElement('span');
 label.textContent=name+" — "+date;
 const done=document.createElement('input');
 done.type='checkbox';
 done.style.width='auto';
 done.style.marginRight='10px';
 done.setAttribute('aria-label','Mark task complete');
 const remove=document.createElement('button');
 remove.textContent='Remove';
 remove.style.marginLeft='10px';
 remove.onclick=()=>item.remove();
 item.append(done,label,remove);
 document.getElementById('tasks').appendChild(item);
 document.getElementById('taskName').value='';
 document.getElementById('taskDate').value='';
}
</script>
